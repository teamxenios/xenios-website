function persistPostHoldOnce(category,stage){

 catch{p.recordPersistence='HOLD_RECORD_FAILED_DO_NOT_RETRY';}
}
async function cancelPostBodyBounded(reader,response){

 try{const operation=reader?reader.cancel():response?.body?.cancel();if(operation)await Promise.race([Promise.resolve(operation).catch(()=>{}),new Promise(resolve=>{timer=setTimeout(resolve,250);})]);}

}
