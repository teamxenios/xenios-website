// Claude independent probe of the corrected founder preview at 516328c (served from the detached review worktree).
// Measures computed image styles, overflow, broken images, ::before/::after overlays on media, src identity per row,
// and the labels used for current/proposed. Loopback only. Writes nothing into the repository.
import { writeFileSync } from 'node:fs';
import { pathToFileURL } from 'node:url';
const LIB = 'C:/xenios-wt/closeout-review/scripts/evidence/lib/';
const { launchChromium } = await import(pathToFileURL(LIB + 'chrome.mjs').href);
const { CdpConnection, PageSession, sleep } = await import(pathToFileURL(LIB + 'cdp.mjs').href);
const { startFounderPreviewServer } = await import(pathToFileURL('C:/xenios-wt/imagery-review-516328c/scripts/product-imagery/serve-founder-preview.mjs').href);
const OUT = process.argv[2];
const server = await startFounderPreviewServer({ port: 0 });
const origin = server.origin ?? server.url ?? `http://127.0.0.1:${server.port ?? server.address?.().port}`;
console.log('SERVER', origin, Object.keys(server));
const views = ['home', 'products', 'featured', 'cards', 'details', 'care', 'held', 'quote', 'pending', 'coming', 'journeys', 'decisions', 'core', 'three-way'];
const paths = [...views.map(v => `/founder-preview/index.html?view=${v}`), '/founder-preview/catalog-review.html?page=1&limit=12',
  '/founder-preview/product-detail.html?id=GRP-0362', '/founder-preview/product-detail.html?id=GRP-0287', '/founder-preview/product-detail.html?id=GRP-0366',
  '/founder-preview/product-detail.html?id=GRP-0063'];
const widths = [1440, 1280, 1024, 834, 768, 430, 390, 375, 360, 320];
const PROBE = `(async () => {
  await Promise.all([...document.images].map(i => i.complete ? 0 : new Promise(r => { i.onload = i.onerror = r; setTimeout(r, 4000); })));
  const imgs = [...document.images].map(i => { const s = getComputedStyle(i); const p = i.parentElement; const ps = getComputedStyle(p);
    const after = getComputedStyle(p, '::after'), before = getComputedStyle(p, '::before');
    const r = i.getBoundingClientRect();
    return { src: i.getAttribute('src'), nat: [i.naturalWidth, i.naturalHeight], box: [Math.round(r.width), Math.round(r.height)], broken: i.complete && i.naturalWidth === 0,
      filter: s.filter, objectFit: s.objectFit, objectPosition: s.objectPosition, mix: s.mixBlendMode, opacity: s.opacity, clip: s.clipPath, mask: s.maskImage || s.webkitMaskImage,
      parentFilter: ps.filter, parentBackdrop: ps.backdropFilter, parentShadow: ps.boxShadow, parentOverflow: ps.overflow, parentRatio: ps.aspectRatio,
      afterContent: after.content, afterBg: after.backgroundImage, afterShadow: after.boxShadow, beforeContent: before.content, beforeBg: before.backgroundImage,
      cls: (p.className || '') + '', label: (p.querySelector('.media-policy-label')?.textContent || '').trim(),
      card: i.closest('[data-canonical-id]')?.dataset.canonicalId || i.closest('[data-canonical]')?.dataset.canonical || null,
      sha: i.closest('[data-asset-sha256]')?.dataset.assetSha256 || null }; });
  const text = document.body.innerText;
  return { title: document.title, scrollW: document.documentElement.scrollWidth, clientW: document.documentElement.clientWidth,
    overflowEls: [...document.querySelectorAll('body *')].filter(e => { const r = e.getBoundingClientRect(); return r.right > document.documentElement.clientWidth + 1 && getComputedStyle(e).position !== 'fixed' && r.width > 0; }).slice(0, 5).map(e => e.tagName + '.' + (e.className + '').slice(0, 40)),
    header: (() => { const h = document.querySelector('header'); if (!h) return null; const r = h.getBoundingClientRect(); return { h: Math.round(r.height), sticky: getComputedStyle(h).position, text: h.innerText.replace(/\\s+/g, ' ').slice(0, 160) }; })(),
    h1: (() => { const h = document.querySelector('h1'); if (!h) return null; const s = getComputedStyle(h); return { t: h.textContent.trim().slice(0, 90), size: s.fontSize, weight: s.fontWeight, family: s.fontFamily.slice(0, 40), bgClip: s.webkitBackgroundClip || s.backgroundClip, color: s.color }; })(),
    bodyBg: getComputedStyle(document.body).backgroundColor, font: getComputedStyle(document.body).fontFamily.slice(0, 40),
    fontsLoaded: [...document.fonts].filter(f => f.status === 'loaded').map(f => f.family + ' ' + f.weight).slice(0, 12),
    currentCount: (text.match(/CURRENT CORE/g) || []).length, proposedCount: (text.match(/PROPOSED XENIOS HEALTH/g) || []).length,
    buttons: [...document.querySelectorAll('a.btn,button.btn,.btn')].slice(0, 6).map(b => { const s = getComputedStyle(b); return { t: b.textContent.trim().slice(0, 30), r: s.borderRadius, bg: s.backgroundColor, h: s.height }; }),
    imgs };
})()`;
const results = [];
for (const width of widths) {
  const chrome = await launchChromium({ chromePath: 'C:/Program Files/Google/Chrome/Application/chrome.exe' });
  const conn = new CdpConnection(chrome.wsUrl); await conn.open();
  const page = await PageSession.create(conn);
  try {
    await page.setViewport({ width, height: 900, deviceScaleFactor: 1, mobile: width < 768 });
    for (const p of (width === 1440 || width === 390 || width === 320) ? paths : paths.slice(0, 15)) {
      try { await page.navigate(origin + p, { maxSettleMs: 8000 }); } catch (e) { await sleep(1500); }
      await sleep(400);
      try { const r = await page.evaluate(PROBE); results.push({ width, path: p, ...r }); }
      catch (e) { results.push({ width, path: p, error: String(e).slice(0, 200) }); }
    }
  } finally { await page.close().catch(() => {}); await conn.close().catch(() => {}); await chrome.close(); }
  console.log('WIDTH DONE', width, results.length);
}
writeFileSync(OUT, JSON.stringify(results, null, 1));
await server.close?.();
process.exit(0);
